wsServer.getUniqueID = function () {
    function s4() {
        return Math.floor((1 + Math.random()) * 0x10000).toString(16).substring(1);
    }		
    return s4() + s4() + '-' + s4();
};

// helper function to convert stream to audio buffer
const getAudioBuffer = async (response) => {
  const reader = response.getReader();
  const chunks = [];
  while (true) {
    const { done, value } = await reader.read();
    if (done) break;
    chunks.push(value);
  }
  const dataArray = chunks.reduce(
    (acc, chunk) => Uint8Array.from([...acc, ...chunk]),
    new Uint8Array(0)
  );
  return Buffer.from(dataArray.buffer);
};
const playback_to_websocket = async (ws, stream) => {  
  if (stream) {
    const reader = stream.getReader();
    let buffer = Buffer.alloc(0);

    while (true) {
      const { done, value } = await reader.read();
      if (done) break;

      const chunk = Buffer.from(value);
      buffer = Buffer.concat([buffer, chunk]);

      while (buffer.length >= 640) {
        ws.send(buffer.subarray(0, 640));
        buffer = buffer.subarray(640);
      }
    }

    // Send any remaining data
    if (buffer.length > 0) {
      ws.send(buffer);
    }
  } else {
    console.error("Error generating audio stream");
  }
};

	//assign an id to this client

			});

			dgConnection.on(LiveTranscriptionEvents.Open, () => {
				console.log("DEEPGRAM CONN OPEN")
				dgConnection.on(LiveTranscriptionEvents.Transcript, async (data) => {
  try {
    const transcript = data.channel.alternatives[0].transcript;
    if (transcript !== '' && transcript !== ' ' && data.is_final) {
      const words = data.channel.alternatives[0].words;
      const message_to_send = {};

      words.forEach(function each(word) {
        if (word.speaker in message_to_send) {
          message_to_send[word.speaker] += " " + word.punctuated_word;
        } else {
          message_to_send[word.speaker] = word.punctuated_word;
        }
      });

      for (const [key, value] of Object.entries(message_to_send)) {
        console.log(key, value);								
        const res = await translate(value, { to: 'en' });
        message_to_send[key] = res.text;
        console.log("Original text language", res.from.language.iso); 
        console.log("Translated text", res.text);
        const text = res.text;

        // If original is not English, synthesize and stream playback
        if (!res.from.language.iso.includes("en")) {
          const response = await deepgram.speak.request(
            { text },
            {
              model: "aura-2-thalia-en",
              encoding: "linear16",
              container: "none",  // Raw output, no header
              sample_rate: 16000,
            }
          );
          const stream = await response.getStream();
          await playback_to_websocket(websocket, stream);  // Stream directly
        }
      }

      console.log("Message: ", message_to_send);
      const to_send = {
        "sessionid": websocket.id,
        "messages": message_to_send
      };
      wsServer.clients.forEach(function each(client) {
        if (client.id === "client_" + websocket.id) {
          client.send(JSON.stringify(to_send));
        }
      });
    }
  } catch (error) {
    console.log("no data", error);
  }
});

    function toggleDeepgram() {
        var button = $('#deepgram_button');

            console.log(`[${new Date().toISOString()}] Initiating Deepgram connection`);
            button.text('Starting Deepgram');
            button.css('background-color', 'Orange');
            $.ajax({

                button.css('background-color', 'darkgreen');

                console.log(`[${new Date().toISOString()}] Deepgram startup failed`);

                button.text('Start Deepgram');
                button.css('background-color', 'darkslategrey');
